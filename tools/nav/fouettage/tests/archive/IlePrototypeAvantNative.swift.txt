    // ⚠️ LE CONTRAT A CHANGÉ LE 05-09 (Kathryn : « partout, à la simple
    // tap ou drag peu importe où sur la display island, je peux faire
    // sortir la pilule — comme les appels iPhone »). Le tap n'ouvre PLUS
    // le player : il SORT la pastille, posée JUSTE SOUS l'île (yRatio 0).
    // Le player s'ouvre depuis la pastille (test03, inchangé).
    func test06_seance_ile_sonTapSortLaPastille_sousLIle() {
        let b = BancPilule(); b.lancerEnSeanceAvecPilule()
        // 1) on l'envoie EN HAUT — la porte volontaire de l'île.
        b.gesteAttendu("cas06a-entre-dans-ile",
                       geste: { b.jeter(b.centrePilule(), dy: -700) },
                       attendu: { $0.dansIle })
        XCTAssertTrue(b.ile.waitForExistence(timeout: 4),
            "l'île n'est pas à l'écran alors que l'état dit qu'on y est")
        // 2) UN TAP SUR L'ÎLE JOUE LA CHAÎNE : baby → grande → la
        //    pastille NORMALE descend à sa place COMMISE (07-09 — plus
        //    jamais « posée sous l'île » : le vol matched rend le voyage
        //    visible, la place basse n'est plus une téléportation).
        b.gesteAttendu("cas06b-tap-ile-sort",
                       geste: { b.taper(b.centreIle()) },
                       attendu: { !$0.dansIle })
        XCTAssertTrue(b.pilule.waitForExistence(timeout: 4),
            "LE TAP DE L'ÎLE NE SORT PAS LA PASTILLE")
    }

    // V2 (07-09) : LE STOP DE LA BABY, à gauche sur la ligne 71 — en

    // du trou). Ce cas prouve que son disque CLAIME le doigt : le

    func test06c_seance_ile_leStopBabyClaqueSansSortir() {

        b.gesteAttendu("cas06c-entre",
                       geste: { b.jeter(b.centrePilule(), dy: -700) },

            "le tap du stop a fait SORTIR la pastille au lieu de stopper")

    // ⚠️ LA CARTE DU TOUCHER (05-09 soir). Kathryn : « je devrais pouvoir
    // la sortir en tapant JUSTE SUR L'ÎLE » — et rien ne se passe sous son
    // doigt. Avant tout remède : mesurer OÙ un tap arrive vraiment.
    // Des taps en grille autour de l'île ; après chacun, la sonde dit
    // si la pastille est sortie (le tap est ARRIVÉ) ou non (zone morte).
    // La carte s'imprime en lignes FOUET-CARTE.
    //
    // ⚠️ GRILLE FINE (06-09) : mesurée AVANT de coder les cotes. VERDICT
    // (21/21, onglet Exercices) : TOUT ARRIVE dès y 55, ailes comprises —
    // le « mort 55-70 » de la carte V1 n'était PAS le système, c'était un
    // mangeur de touchers de la HOME (menu de fumée, corrigé 5c21bd3) ;
    // la V1 se mesurait sur la home, la capsule n'y vit plus (gate du
    // Foyer). Le vrai bloc mort = la bande du trou (y ≤ ~50).
    // ⚠️ LEÇON du 06-09 : une carte dont une case OUVRE une UI (le stop →
    // sa card) empoisonne les cases suivantes — le stop se juge par
    // test06c, jamais par la carte. Sur la BABY (07-09), son disque vit
    // à (midX−50, 71) : aucune colonne de la grille ne le touche.
    func test99_carteDuToucher_autourDeLile() {
        let b = BancPilule(); b.lancerEnSeanceAvecPilule()
        let ecran = b.app.frame
        let xs: [(String, CGFloat)] = [("aile-G", ecran.midX - 108),
                                       ("centre", ecran.midX),
                                       ("aile-D", ecran.midX + 90)]
        let ys: [CGFloat] = [55, 58, 65, 70, 72, 75, 80]
        // On commence DANS l'île.
        b.gesteAttendu("carte-entre",
                       geste: { b.jeter(b.centrePilule(), dy: -700) },
                       attendu: { $0.dansIle })
        for y in ys {
            for (nom, x) in xs {
                guard b.attendre(4, { $0.dansIle }) else {
                    print("FOUET-CARTE y=\(Int(y)) \(nom) : PAS TESTÉ (retour d'île manqué)")
                    continue
                }
                b.taper(b.pointEcran(x: x, y: y))
                Thread.sleep(forTimeInterval: 1.2)
                let sorti = !(b.etat()?.dansIle ?? true)
                print("FOUET-CARTE y=\(Int(y)) \(nom) x=\(Int(x)) : \(sorti ? "ARRIVE — la pastille sort" : "mort")")
                if sorti {
                    // on la renvoie dans l'île pour le point suivant
                    b.gesteAttendu("carte-rentre",
                                   geste: { b.jeter(b.centrePilule(), dy: -700) },
                                   attendu: { $0.dansIle })
                }
            }
        }
    }

    func test10_ile_leCadreEnglobeLeCapteurEtDegageLeTitre() {
        let b = BancPilule(); b.lancerEnSeanceAvecPilule()
        b.gesteAttendu("cadre-entre",
                       geste: { b.jeter(b.centrePilule(), dy: -700) },
                       attendu: { $0.dansIle })
        XCTAssertTrue(b.ile.waitForExistence(timeout: 4))
        Thread.sleep(forTimeInterval: 1)
        let cadre = b.ile.frame
        XCTAssertLessThan(cadre.minY, 11, "le cadre doit entourer le capteur dès le repos")
        XCTAssertGreaterThan(cadre.maxY, 66, "le contenu doit rester dans le bloc")
        let titre = b.app.staticTexts["Exercices"].firstMatch
        XCTAssertTrue(titre.exists)
        XCTAssertLessThanOrEqual(cadre.maxY, 72.5,
                                 "la capsule doit tenir au-dessus du dessin des lettres")
        XCTAssertLessThan(titre.frame.minY, 65,
                          "la page doit conserver sa hauteur d'origine")
        let capture = XCTAttachment(screenshot: b.app.screenshot())
        capture.name = "ile-integree-exercices"; capture.lifetime = .keepAlways
        add(capture)
    }
